 /* finding greatest of 3 numbers*/
 #include<stdio.h>
 int main()
 { 
     int a,b,c;
     printf("enter the values of a,b,c:");
     scanf("%d%d%d",&a,&b,&c);
     if(a>=b&&a>=c)
     printf("%d is greatest",a);
     if(b>=a&&b>=c)
     printf("%d is greatest",b);
     if(c>=a&&c>=b)
     printf("%d is greatest",c);
     else 
     printf("invalid....");
     return 0;
 }
